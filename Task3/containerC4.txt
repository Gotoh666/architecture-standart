Person(client, "Клиент банка", "")
Person(userfront, "Сотрудник фронт офиса", "")
note right of userfront
На время трансформации работает в двух интерфейсах
В старом АБС и в новом фронтенде
end note
Person(userBack, "Сотрудник бэк офиса", "")
note right of userBack
На время трансформации работает в двух интерфейсах
В старом АБС и в новом фронтенде
end note
Person(userCall, "Сотрудник кол-центра", "")
Person(userExtCall, "Сотрудник партнерского кол-центра", "")

System_Ext(teleoperator, "Телеком-оператор")
System_Ext(extcall, "Система партнёрского кол-центра")

System_Boundary(system, "Система банка (Только депозиты)"){

Container(site, "Сайт", "PHP, React")
Container(internetbank, "Интернет-банк", "ASP.NET MVC 4.5, MS SQL")
Container(abs, "ABS", "Delphi, Oracle")
Container(call, "Система кол-центра", "CRM system, Java, React, PostgreSQL")
Container(smsgateway, "СМС-шлюз телеком-оператора", "")

Container(bff, "BFF", "Asp NET Core, Yarp")
note right of bff
Шлюз между существюущими системами, новыми микросервисами и АБС
Распределяет запросы, балансирует нагрузку
В будущем возьмет на себя роль BFF
end note

Container(frontend, "Frontend", "react.js")
note right of frontend
Простое веб приложение для работы с депозитами, ставками
В будущем будет развиваться в фронтенде всей системы банка для сотрудников
end note

Container(deposit_service, "Deposit Service", "ASP.NET Core")
Container(rates_service, "Rates Service", "ASP.NET Core")
note right of rates_service
Сервис ставок, расчитывает и хранит ставки.
Актуальные ставки кешируются в памяти.
end note

Container(acl_gateway, "ABS Gateway / Anti-Corruption Layer", "ASP.NET Core")
note right of acl_gateway
Шлюз между Deposit Service и ABS.
Трансформирует модели данных для новой системы
end note

ContainerDb(deposit_service_db, "Deposit Service DB", "MS SQL")
ContainerDb(abs_db, "ABS DB", "Oracle")
ContainerDb(call_db, "Call DB", "PostgreSQL")
ContainerDb(rates_service_db, "Rates Service DB", "MS SQL")

}

Rel(userfront, abs, "Работает с банковскими операциями", "")
Rel(userBack, abs, "Работает с банковскими операциями", "")
Rel(userfront, frontend, "Работает с депозитами", "")
Rel(userBack, frontend, "Работает с депозитами", "")

Rel(userCall, call, "Создает заявки на депозит", "")
Rel(userExtCall, extcall, "Создает заявки на депозит", "")

Rel(client, site, "Работает с депозитами", "")
Rel(client, internetbank, "Работает с депозитами", "")

Rel(smsgateway, teleoperator, "Отправка смс", "")
Rel(abs, smsgateway, "Отправляет смс", "")

Rel(site, bff, "Работает с депозитами", "")
Rel(internetbank, bff, "Работает с депозитами. Проводит платежи, открывает дебетовые счета", "")
Rel(call, bff, "Передает заявки", "")
Rel(extcall, bff, "Создает заявки на депозит", "")

Rel(bff, deposit_service, "Запросы к сервису депозитов", "")
Rel(bff, rates_service, "Запросы к сервису ставок", "")

Rel(deposit_service, acl_gateway, "Операции по депозитам", "")
Rel(acl_gateway, abs, "Интеграция с legacy", "")

Rel(deposit_service, deposit_service_db, "CRUD по депозитам", "")
Rel(abs, abs_db, "", "")
Rel(call, call_db, "", "")
Rel(rates_service, rates_service_db, "", "")

Rel(deposit_service, rates_service, "Получает ставки", "")
Rel(frontend, bff, "Получает данные по депозитам, создает депозит, Получает данные по ставкам, обновляет ставки", "")

@enduml
